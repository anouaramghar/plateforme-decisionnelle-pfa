```python
# ml_service/routers/metrics.py
from fastapi import APIRouter, Depends, HTTPException, Header
from typing import Optional
import joblib, json, os
from pathlib import Path
from sklearn.metrics import roc_auc_score, f1_score, precision_score, recall_score
import pandas as pd

router = APIRouter(prefix="/metrics", tags=["metrics"])

INTERNAL_TOKEN = os.getenv("ML_INTERNAL_TOKEN", "")
MODELS_DIR = Path(os.getenv("ML_MODELS_DIR", "/app/saved_models"))

def _verify_token(x_internal_token: Optional[str] = Header(default=None, alias="X-Internal-Token")):
    if not INTERNAL_TOKEN or x_internal_token != INTERNAL_TOKEN:
        raise HTTPException(status_code=401, detail="invalid token")

def _load_eval_set():
    """Held-out test set persisted at training time as parquet + numpy arrays.
    Falls back to synthetic data so the endpoint never 500s in dev."""
    eval_path = MODELS_DIR / "eval_set.parquet"
    if not eval_path.exists():
        return None
    df = pd.read_parquet(eval_path)
    X = df.drop(columns=["y_true"])
    y = df["y_true"].values
    return X, y

@router.get("", dependencies=[Depends(_verify_token)])
def get_metrics():
    meta_path = MODELS_DIR / "metadata.json"
    if meta_path.exists():
        meta = json.loads(meta_path.read_text())
    else:
        meta = {"model_version": "unknown", "trained_at": None}

    eval_data = _load_eval_set()
    if eval_data is None:
        # No held-out set → return last-known metrics from metadata only.
        return {
            "auc": meta.get("auc", 0.0),
            "f1": meta.get("f1", 0.0),
            "precision": meta.get("precision", 0.0),
            "recall": meta.get("recall", 0.0),
            "n_samples": meta.get("n_samples", 0),
            "model_version": meta["model_version"],
            "trained_at": meta["trained_at"],
            "source": "metadata",
        }

    X, y = eval_data
    model = joblib.load(MODELS_DIR / "risque_model.joblib")
    probs = model.predict_proba(X)[:, 1]
    preds = (probs >= 0.5).astype(int)
    return {
        "auc": float(roc_auc_score(y, probs)),
        "f1": float(f1_score(y, preds, zero_division=0)),
        "precision": float(precision_score(y, preds, zero_division=0)),
        "recall": float(recall_score(y, preds, zero_division=0)),
        "n_samples": int(len(y)),
        "model_version": meta["model_version"],
        "trained_at": meta["trained_at"],
        "source": "computed",
    }
```

// GET: api/predictions/metrics — proxies the ML service's /metrics so the
// frontend doesn't need to know about ML_INTERNAL_TOKEN.
[HttpGet("metrics")]
public async Task<ActionResult<MlMetricsDto>> GetMetrics()
{
    try
    {
        var mlApiUrl = _configuration["ML_API_URL"] ?? "http://ml-service:8000";
        var client = _httpClientFactory.CreateClient("MLService");
        var request = new HttpRequestMessage(HttpMethod.Get, $"{mlApiUrl}/metrics");
        request.Headers.Add("X-Internal-Token", _configuration["ML_INTERNAL_TOKEN"]);
        var response = await client.SendAsync(request);
        if (!response.IsSuccessStatusCode)
            return StatusCode((int)response.StatusCode, new { message = "ML metrics unavailable" });

        var body = await response.Content.ReadAsStringAsync();
        var dto = System.Text.Json.JsonSerializer.Deserialize<MlMetricsDto>(body, new System.Text.Json.JsonSerializerOptions
        {
            PropertyNameCaseInsensitive = true,
        });
        return dto!;
    }
    catch (Exception ex)
    {
        _logger.LogWarning(ex, "ML metrics fetch failed");
        return StatusCode(StatusCodes.Status502BadGateway, new { message = "ML service unreachable" });
    }
}
```

// Compare against the same metrics computed on Notes from the previous
// semester (S1 if currently S2, or last academic year's S2 if currently S1).
var (currentSem, prevAnnee, prevSem) = ResolvePreviousPeriod();

var prevStudents = await _context.Etudiants
    .AsNoTracking()
    .Select(e => new
    {
        Moyenne = (decimal?)e.Notes!
            .Where(n => n.NoteFinal.HasValue && n.Annee == prevAnnee && n.Semestre == prevSem)
            .Average(n => n.NoteFinal),
    })
    .ToListAsync();

var prevWithMoy = prevStudents.Where(s => s.Moyenne.HasValue).ToList();
var prevMoy = prevWithMoy.Count > 0
    ? Math.Round(prevWithMoy.Average(s => s.Moyenne!.Value), 2)
    : 0m;
var prevReussite = prevWithMoy.Count > 0
    ? Math.Round((decimal)prevWithMoy.Count(s => s.Moyenne!.Value >= 10m) / prevWithMoy.Count * 100m, 1)
    : 0m;

// Alerts delta = active alerts now vs active alerts 7 days ago. Active means
// not resolved as of that timestamp.
var weekAgo = DateTime.UtcNow.AddDays(-7);
var prevActiveAlerts = await _context.Alertes
    .CountAsync(a => a.CreeLe < weekAgo && (a.ResolueeLe == null || a.ResolueeLe >= weekAgo));

// Student count delta = % growth in CreeLe last 30 days vs prior 30 days.
var since30 = DateTime.UtcNow.AddDays(-30);

var newLast30 = await _context.Etudiants.CountAsync(e => e.CreeLe >= since30);
var newPrev30 = await _context.Etudiants.CountAsync(e => e.CreeLe < since30 && e.CreeLe >= since60);

private static (string CurrentSem, string PrevAnnee, string PrevSem) ResolvePreviousPeriod()
{
    var now = DateTime.UtcNow;
    var startYear = now.Month >= 9 ? now.Year : now.Year - 1;
    // School year flips Sep 1; S1 runs Sep–Jan, S2 runs Feb–Jul roughly.
    var currentSem = now.Month >= 2 && now.Month <= 8 ? "S2" : "S1";
    if (currentSem == "S2")
    {
        return (currentSem, $"{startYear}/{startYear + 1}", "S1");
    }
    else
    {
        return (currentSem, $"{startYear - 1}/{startYear}", "S2");
    }
}
```

            string? message = null, int? userId = null, string? userName = null)
        {
            try
            {
                _context.AuditEntries.Add(new AuditEntry
                {
                    UtilisateurId  = userId,
                    UtilisateurNom = userName ?? "Système",
                    Action         = action,
                    EntityType     = entityType ?? "",
                    EntityId       = entityId,
                    Message        = message,
                    CreeLe         = DateTime.UtcNow,
                });
                await _context.SaveChangesAsync();
            }
            catch (Exception ex)
            {
                // Audit must never block the caller. Swallow + log.

private static (string Annee, string? Semestre) ParsePeriode(string periode)
{
    // "Semestre 2 · 2025/2026" → ("2025/2026", "S2")
    // "Année 2024/2025"        → ("2024/2025", null)
    // Fallback: current academic year, no semester filter.
    var anneeMatch = System.Text.RegularExpressions.Regex.Match(periode, @"\d{4}/\d{4}");
    var annee = anneeMatch.Success ? anneeMatch.Value : CurrentAcademicYear();
    string? sem = null;
    if (periode.Contains("Semestre 1")) sem = "S1";

}

```csharp
[HttpPost("retrain")]
[Authorize(Roles = "Admin")]
public async Task<ActionResult> Retrain()
{
    try
    {
        var mlApiUrl = _configuration["ML_API_URL"] ?? "http://ml-service:8000";
        var client = _httpClientFactory.CreateClient("MLService");
        var req = new HttpRequestMessage(HttpMethod.Post, $"{mlApiUrl}/retrain");
        req.Headers.Add("X-Internal-Token", _configuration["ML_INTERNAL_TOKEN"]);
        var resp = await client.SendAsync(req);
        var body = await resp.Content.ReadAsStringAsync();
        if (!resp.IsSuccessStatusCode)
            return StatusCode((int)resp.StatusCode, body);

        await _audit.LogAsync("MlRetrain", "ML",
            message: body, userId: ResolveUserId(), userName: ResolveUserName());
        return Content(body, "application/json");
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Retrain failed");
        return StatusCode(StatusCodes.Status502BadGateway, new { message = "ML service unreachable" });
    }
}
```
